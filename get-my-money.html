    <script>

        const words = ["💰 发财啦", "☕ 奶茶到手", "🍟 麦门永存", "🏠 房租攒够", "👔 下班倒数", "✨ 值得奖励", "💼 拿捏老板", "🌈 自由在望"];

        const moneyDisplay = document.getElementById('moneyDisplay');

        const dailyGoalDisplay = document.getElementById('dailyGoal');

        const progressDisplay = document.getElementById('progressPercent');

        const moneyContainer = document.getElementById('money-container');


        function createFloatingText() {

            const word = words[Math.floor(Math.random() * words.length)];

            const el = document.createElement('div');

            el.className = 'floating-word';

            el.innerText = word;

            el.style.left = Math.random() * 60 + 20 + '%';

            el.style.color = `hsl(${Math.random() * 360}, 70%, 70%)`;

            moneyContainer.appendChild(el);

            setTimeout(() => el.remove(), 2000);

        }


        function calculate() {

            const salary = parseFloat(document.getElementById('salaryInput').value) || 0;

            const endTimeStr = document.getElementById('endTimeInput').value;

            

            const now = new Date();

            const start = new Date();

            start.setHours(9, 0, 0, 0); // 默认9点上班

            

            const end = new Date();

            const [h, m] = endTimeStr.split(':');

            end.setHours(parseInt(h), parseInt(m), 0, 0);


            // 算法：月薪 / 21.75天 = 日薪

            const dailyPay = salary / 21.75;

            dailyGoalDisplay.innerText = `¥${dailyPay.toFixed(2)}`;


            if (now < start) {

                moneyDisplay.innerText = "¥0.0000";

                progressDisplay.innerText = "0.0%";

                return;

            }


            if (now > end) {

                moneyDisplay.innerText = `¥${dailyPay.toFixed(2)}`;

                progressDisplay.innerText = "100%";

                return;

            }


            // 计算每秒多少钱

            const totalWorkSeconds = (end - start) / 1000;

            const payPerSec = dailyPay / totalWorkSeconds;

            const passedSeconds = (now - start) / 1000;

            const earned = passedSeconds * payPerSec;


            moneyDisplay.innerText = `¥${earned.toFixed(4)}`;

            progressDisplay.innerText = `${((earned / dailyPay) * 100).toFixed(1)}%`;


            // 1% 概率产生漂浮文字

            if (Math.random() > 0.985) {

                createFloatingText();

            }

        }


        // 每 100 毫秒更新一次，保证丝滑

        setInterval(calculate, 100);

    </script>
